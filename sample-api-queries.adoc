= Use Digital Advisor GraphQL API example queries
:icons: font
:imagesdir: ./media/

[.lead]
After you've generated your tokens and learned how to create a GraphQL request, you can use the example queries to retrieve various types of data from the GraphQL API for Digital Advisor.

Before you use the GraphQL query examples, review the following information:

* Replace `$variableName` with actual values or use GraphQL variables.
* Use cursor-based pagination for better performance with large datasets.
* Adjust the `pageSize` based on your application's needs.
* Include only the fields you need to optimize query performance.
* Consider using fragments for reusable field sets.
* Test your queries in the GraphQL Playground or a similar tool before implementation.

For more example GraphQL queries, refer to the link:https://studio.apollographql.com/public/ActiveIQ-Graph-Prd-API/variant/current/explorer?collectionId=3c60f539-811a-4494-bfa6-55240b953cbf&focusCollectionId=3c60f539-811a-4494-bfa6-55240b953cbf[Apollo Studio explorer collections^].

.Python query example

.Show query
[%collapsible]
====
[source,python]
----
import requests
import json

def get_sites():
    graphql_endpoint = 'https://gql.aiq.netapp.com/graphql'
    access_token = 'YOUR_ACCESS_TOKEN'
    
    # GraphQL query
    query = """
        query($customerId: String, $after: String) {
          sites(customerId: $customerId, after: $after, pageSize: 10) {
            cursor
            sites {
              id
              name
            }
          }
        }
    """
    
    # Variables
    variables = {
        'customerId': 'your-customer-id',
        'after': None
    }
    
    # Headers
    headers = {
        'Content-Type': 'application/json',
        'Authorization': f'Bearer {access_token}'
    }
    
    # Request body
    payload = {
        'query': query,
        'variables': variables
    }
    
    try:
        # Send request
        response = requests.post(
            graphql_endpoint,
            headers=headers,
            json=payload
        )
        
        # Output
        print(f'Status Code: {response.status_code}')
        print('Response Body:')
        print(json.dumps(response.json(), indent=2))
        
        return response.json()
    except Exception as e:
        print(f'Error: {e}')

if __name__ == '__main__':
    get_sites()

.Java query example
--

.Show query
[%collapsible]
====
[source,java]
----
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import org.json.JSONObject;

public class GraphQLSitesQuery {
    public static void main(String[] args) throws Exception {
        String graphqlEndpoint = "https://gql.aiq.netapp.com/graphql";
        String accessToken = "YOUR_ACCESS_TOKEN";
        
        String query = """
            query($customerId: String, $after: String) {
              sites(customerId: $customerId, after: $after, pageSize: 10) {
                cursor
                sites {
                  id
                  name
                }
              }
            }
            """;
        
        JSONObject variables = new JSONObject();
        variables.put("customerId", "your-customer-id");
        variables.put("after", JSONObject.NULL);
        
        JSONObject requestBody = new JSONObject();
        requestBody.put("query", query);
        requestBody.put("variables", variables);
        
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(graphqlEndpoint))
            .header("Content-Type", "application/json")
            .header("Authorization", "Bearer " + accessToken)
            .POST(HttpRequest.BodyPublishers.ofString(requestBody.toString()))
            .build();
        
        HttpResponse<String> response = client.send(request, 
            HttpResponse.BodyHandlers.ofString());
        
        System.out.println("Status Code: " + response.statusCode());
        System.out.println("Response Body:");
        System.out.println(response.body());
    }
}

const fetch = require('node-fetch'); // or use axios

async function getSites() {
    const graphqlEndpoint = 'https://gql.aiq.netapp.com/graphql';
    const accessToken = 'YOUR_ACCESS_TOKEN';
    
    // GraphQL query
    const query = `
        query($customerId: String, $after: String) {
          sites(customerId: $customerId, after: $after, pageSize: 10) {
            cursor
            sites {
              id
              name
            }
          }
        }
    `;
    
    // Variables
    const variables = {
        customerId: 'your-customer-id',
        after: null
    };
    
    try {
        const response = await fetch(graphqlEndpoint, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${accessToken}`
            },
            body: JSON.stringify({
                query: query,
                variables: variables
            })
        });
        
        const data = await response.json();
        
        // Output
        console.log('Status Code:', response.status);
        console.log('Response Body:');
        console.log(JSON.stringify(data, null, 2));
        
        return data;
    } catch (error) {
        console.error('Error:', error);
    }
}

getSites();

=== Track all purchased systems

You can use this query to identify which systems are sending telemetry data. This automatically detects gaps in telemetry and for automatic case generation.

.Show example
[%collapsible]
====
[source,graphql]
----
query($customerId: String, $after: String) {
  # Get a summary of AutoSupport status across the fleet
  summary(customerId: $customerId) {
    system
    countByField(countBy: AUTOSUPPORT_STATUS) {
      fieldName
      counts {
        system
      }
    }
  }
  
  # Fetch systems that are not sending telemetry data
  systems (customerId: $customerId, autoSupportStatus: [OFF, NA, DECLINE, RSS], after: $after, pageSize: 10) {
    cursor
    totalCount
    systems {
      serialNumber
      systemId
      autoSupportConfig {
        autoSupportStatus
      }
    }
  }
}

=== Discover and monitor risk

You can use this query to discover and monitor security, upgrade, best practice, configuration, community, and AI issues.

.Show example
[%collapsible]
====
[source,graphql]
----
query($customerId: String, $after: String) {
  # Fetch count of risks for critical and high severity issues
  risksCount (
    customerId: $customerId,
    filter: {
      severity: [
        CRITICAL
        HIGH
      ]
    }
  ) {
    riskCount
    systemCount
    actionCount
  }
  # Fetch risks for critical and high severity issues
  risks(
    customerId: $customerId,
    after: $after,
    pageSize: 10,
    filter: {
      severity: [
        CRITICAL
        HIGH
      ]
    }
  ) {
    cursor
    risks {
      riskId
      impactArea
      severity
      shortName
      riskDetail
      mitigationAction
      potentialImpact
      riskInstances {
        system {
          serialNumber
          systemId
          platformType
        }
      }
    }
  }
}
